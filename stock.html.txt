    });

function safeSelectHero(id, color, bulletColor) {
    if(!player) return;
    player.heroId = id; player.heroColor = color; player.heroBulletColor = bulletColor;
    player.perks.crit = (id === 'leon');
    player.fireSpeedBonus = (id === 'chloe') ? 90 : 0;
    player.chipLevels.DRONE = (id === 'sigma') ? 1 : 0;
    safeEnterHub();
}

function safeEnterHub() {
    gameState = 'HUB'; boss = null; isRewinding = false; metaTriggered = false;
    document.getElementById('metaAlert').style.display = 'none';
    document.getElementById('rewindIndicator').style.display = 'none';
    
    document.querySelectorAll('.rpg-screen').forEach(s => s.classList.add('hidden'));
    document.getElementById('combatHud').classList.add('hidden');
    document.getElementById('screenHub').classList.remove('hidden');
    document.getElementById('hubPoints').innerText = player.points;
    
    const chipTypes = ['OVERCLOCK', 'HYPERDRIVE', 'DRONE'];
    const baseCosts = { OVERCLOCK: 12, HYPERDRIVE: 4, DRONE: 5 };
    chipTypes.forEach(type => {
        let lvl = player.chipLevels[type];
        let nextCost = Math.floor(baseCosts[type] * Math.pow(1.8, lvl));
        document.getElementById(`lvl${type}`).innerText = `Lv.${lvl}`;
        document.getElementById(`cost${type}`).innerText = nextCost;
        document.getElementById(`btn${type}`).disabled = player.points < nextCost;
    });
}

function safeBuyChip(type, baseCost) {
    let lvl = player.chipLevels[type]; 
    let currentCost = Math.floor(baseCost * Math.pow(1.8, lvl));
    if(player.points >= currentCost) {
        player.points -= currentCost; player.chipLevels[type] = lvl + 1;
        player.dmgMultiplier = 1.0 + player.chipLevels.OVERCLOCK * 1.20;
        let baseSpeed = (player.heroId === 'chloe') ? 90 : 0;
        player.fireSpeedBonus = baseSpeed + player.chipLevels.HYPERDRIVE * 60;
        let baseHp = (player.heroId === 'sigma') ? 130 : 100;
        player.maxHp = Math.max(50, baseHp - player.chipLevels.HYPERDRIVE * 3); 
        safeEnterHub();
    }
}

function safeStartCombat() {

    document.getElementById('rewindIndicator').style.display = 'block';
    
    let baseHp = (player.heroId === 'sigma') ? 130 : 100;
    player.maxHp = Math.max(50, baseHp - player.chipLevels.HYPERDRIVE * 3);
    player.hp = player.maxHp;
    player.killCount = 0; boss = null; historyLog = []; playerTrails = [];
    bullets = []; enemies = []; particles = []; dmgs = []; rewindCooldown = 0;
    lastSpawn = Date.now(); lastShot = Date.now();
    updateHUD();
}

function updateHUD() {
    document.getElementById('hudFloor').innerText = player.floor < 10 ? `0${player.floor}` : player.floor;
    document.getElementById('hudScore').innerText = player.currentScore;
    document.getElementById('hudHp').innerText = Math.max(0, Math.floor((player.hp / player.maxHp) * 100));
    
    let perksDiv = document.getElementById('hudPerks'); perksDiv.innerHTML = '';
    if(player.perks.split) perksDiv.innerHTML += `<span class="perk-badge">🧬 分裂</span>`;
    if(player.perks.lifesteal) perksDiv.innerHTML += `<span class="perk-badge">🩸 護盤</span>`;

    if(player.perks.crit) perksDiv.innerHTML += `<span class="perk-badge">🎯 內線</span>`;
}

function triggerHit(x, y, text, color, shake = 6) {
    screenShake = shake; 
    if(dmgs.length < 25) {
        dmgs.push({x: x - 20, y: y - 15, text: text, life: 35, color: color, size: text.includes('!') ? 18 : 13});
    }
}

function spawnExplosionParticles(x, y, color, count = 15, speedScope = 7) {
    for(let i=0; i<count; i++) {
        let angle = Math.random() * Math.PI * 2; 
        let speed = (Math.random() * 0.6 + 0.4) * speedScope;
        particles.push({
            x: x, y: y, vx: Math.cos(angle) * speed, vy: Math.sin(angle) * speed,
            life: Math.random() * 25 + 10, color: color, size: Math.random() * 4 + 1.5
        });
    }
}

    let heroNames = { leon: "歐印流 Leon", chloe: "當沖女王 Chloe", sigma: "量化巨鯨 AI" };
    let currentHeroName = heroNames[player.heroId] || "散戶戰機";
    
    let localData = localStorage.getItem('cyber_taiwan_rich_list_v2');
    let list = localData ? JSON.parse(localData) : [];
    
    list.push({ name: nameInput, hero: currentHeroName, score: player.currentScore });
    list.sort((a, b) => b.score - a.score);
    list = list.slice(0, 5); // 只取前五名
    
    localStorage.setItem('cyber_taiwan_rich_list_v2', JSON.stringify(list));
    
    let tbody = document.getElementById('leaderboardBody'); tbody.innerHTML = "";
    let medals = ["🥇", "🥈", "🥉", "4", "5"];
    list.forEach((item, idx) => {
        let row = document.createElement('tr');
        row.innerHTML = `<td>${medals[idx]}</td><td style="color:#39ff14;font-weight:bold;">${item.name}</td><td style="color:#00ffff;">${item.hero}</td><td style="color:#f59e0b;font-weight:bold;">$${item.score.toLocaleString()}</td>`;
        tbody.appendChild(row);
    });
    
    document.getElementById('screenGameOver').classList.add('hidden');
    document.getElementById('screenLeaderboard').classList.remove('hidden');
}
